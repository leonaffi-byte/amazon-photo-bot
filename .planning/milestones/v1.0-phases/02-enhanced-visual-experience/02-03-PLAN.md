def to_product_info_list(self) -> list:
    # For multi-product results:

        bbox_raw = p.get("bbox")
        bbox = tuple(bbox_raw) if bbox_raw and len(bbox_raw) == 4 else None
        result.append(ProductInfo(..., bbox=bbox))

    #   uses self.bbox directly (set from first product in products_raw)
